CONSTANT database_url : ":///sqlite3..."
CONSTANT credentials = "username, password"

/* ======================================                      STUDENT MANAGAEMENT                      ======================= */

PROCEDURE StudentManagement()

	DECLARE operation : INTEGER
	
	WHILE TRUE DO
	
		//Show all the students
		CALL ShowAllStudent()
		
		//Then, ask for CRUD operation
		INPUT("
		Please enter a number
		1.Add new student
		2.Delete student
		3.Update student
		4.Search student
		5.exit
		", operation)
		
		CASE OF operation
			1 : CALL AddStudent()
			2 : CALL DeleteStudent()
			3 : CALL UpdateStudent()
			4 : CALL SearchStudent()
			5 : BREAK
			
	ENDWHILE
	
ENDPROCEDURE

// =====================      Operation Selection     ======================

PROCEDURE ShowAllStudent()
	CONSTANT connection = ConnectToDatabase(database_url, credential)
	CONSTANT query = " """ SELECT * FROM Students """ "
	DECLARE result : ARRAY OF DICTIONARY
	IF connection == FALSE THEN
		OUTPUT "connection error: " + connection.error
	ENDIF
	result = EXECUTE query
	FOR EACH record in result
		FOR EACH (key, value) IN record
			OUTPUT key, value
		ENDFOR
	NEXT record
	
ENDPROCEDURE

FUNCTION checkStudentExist(name : STRING, class : INTEGER) RETURNS BOOLEAN
	DECLARE result : ARRAY OF DICTIONARY
	CONSTANT query = " """SELECT studentName FROM Students WHERE studentName = ? AND class = ?""",name, class"
	CONSTANT connection = ConnectToDatabase(database_url, credential)
	IF connection == FALSE Then
		OUTPUT "connection error " + connection.error
		RETURN FALSE
	ENDIF
	
	SET result = EXECUTE query
	IF COUNT(result) > 0 THEN
		RETURN TRUE
	ELSE
		RETURN FALSE
	ENDIF
	
ENDFUNCTION

PROCEDURE AddStudent()
	DECLARE studentName : STRING
	DECLARE age : INT
	
	//Ask for name and age
	INPUT("Please enter name and age", studentName, age)
	
	//Validate name and age
	WHILE NOT IsAlphanumeric(studentName) OR NOT IsNumeric(age) DO
		INPUT("Please enter alphanumeric name and age", studentName, age)
	ENDWHILE
	
	//Adding new student to Student table && Output error if found
	DB_AddStudent(studentName, age)
		
ENDPROCEDURE

PROCEDURE UpdateStudent()

	DECLARE class : INTEGER

PROCEDURE SearchStudent()

		OUTPUT "Connection erorr: " + connection.error

	ENDIF
	
	CONSTANT query = " """
		INSERT INTO Students (classID, name, age)
		VALUES (-1, ?, ?); """, studentName, age "

	//Execute the query

ENDPROCEDURE

PROCEDURE DB_SearchStudent(studentName : STRING)
