PROCEDURE AddStudentToSubject(studentName : STRING, subjectName : STRING)
	
	//connection to DB
	SET connection = ConnectionToDatabase()
	IF connection == FALSE THEN
		OUTPUT "connection error" + connection.error
		BREAK
	ENDIF
	
	//name validation
	IF NOT checkStudentExist(studentName) OR NOT checkSubjectExist(subjectName) THEN
		OUTPUT "Student or subject name not exist"
		BREAK
	ENDIF
	
	CONSTANT query = " """
		INSERT INTO Enrollments(studentID, subjectID)
		VALUES (
			SELECT studentID FROM Students WHERE studentName = ?
		,
			SELECT subjectID FROM Subjects WHERE subjectName = ?
		) """, studentName, subjectName "
	
	EXECUTE query
	OUTPUT "successfully added" + studentName "To" + subjectName"
	
ENDPROCEDURE

PROCEDURE SearchSubjectFromStudent(studentName : STRING)
	DECLARE result = ARRAY OF DICTIONARY
	
	//connection to DB
	SET connection = CALL ConnectToDatabase(username, credentials)
	IF connection == FALSE THEN
		OUTPUT "connection error" + connection.error
		BREAK
	ENDIF
	
	CONSTANT query = " """
		SELECT subjectName 
		FROM Subjects 
		JOIN Students ON Students.studentID = Subjects.subjectID
		WHERE studentName = ?
		""", studentName "
		
	SET results = EXECUTE query
	
	OUTPUT studentName "chosen subjects: "
	FOR EACH row IN result
		OUTPUT row["subjectName"]
	ENDFOR
	
	//close connection
	CALL CloseDatabaseConnection()
	
ENDPROCEDURE

PROCEDURE RemoveSubjectOfStudent()
	DECLARE studentName : STRING
	DECLARE subjectName : STRING
	
	CONSTANT query = " """
	DELETE FROM Enrollments 
	JOIN Students ON Enrollments.studentID = Students.studentID
	JOIN Subjects ON Enrollments.subjectID = Subjects.subjectID
	WHERE studentName = ? 
		AND subjectName = ?;
	""", studentName, subjectName "
	
	INPUT("Deleting subject from a student, please enter student and subject", studentName, subjectName)
	
	IF NOT Alphanumeric(studentName) OR NOT Alphanumeric(subjectName) THEN
		OUTPUT "Student name or subject name should contain only letter and number"
		BREAK
	ENDIF
	
	IF NOT checkStudentExist(studentName) OR NOT checkSubjectExist(subjectName) THEN
		OUTPUT "Student or subject does not exist"
		BREAK
	ENDIF
	
	SET connection = CALL ConnectToDatabase(username, credentials)
	IF connection == FALSE THEN
		OUTPUT "connection fail " + connection.error
		BREAK
	ENDIF
	
	EXECUTE query
	OUTPUT "Sucessfully deleted student from a subject"
	
ENDPROCEDURE
