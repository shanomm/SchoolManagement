CONSTANT database_url : ":///sqlite3..."
CONSTANT credentials = "username, password"

IMPORT IsValidName, checkClassExist, checkClassNameExist FROM DB_util.txt

PROCEDURE ClassManagement()

	DECLARE operation : INTEGER

	WHILE TRUE DO

		CALL ShowAllClass()

		INPUT("
		Please enter a number
		1.Add new class
		2.Delete class
		3.Update class
		4.exit
		", operation)

		CASE OF operation
			1 : CALL AddClass()
			2 : CALL DeleteClass()
			3 : CALL UpdateClass()
			4 : RETURN
			OTHERWISE : OUTPUT "Invalid option, please enter 1-4"
		ENDCASE

	ENDWHILE

ENDPROCEDURE

// =====================      Operation Selection     ======================

PROCEDURE ShowAllClass()
	DECLARE result : ARRAY OF DICTIONARY
	CONSTANT connection = ConnectToDatabase(database_url, credentials)
	IF connection == FALSE THEN
		OUTPUT "Connection error: " + connection.error
		RETURN
	ENDIF
	result = EXECUTE "SELECT * FROM Class"
	CALL CloseDatabaseConnection()
	IF COUNT(result) == 0 THEN
		OUTPUT "No classes found"
		RETURN
	ENDIF
	FOR EACH record IN result
		FOR EACH (key, value) IN record
			OUTPUT key, value
		NEXT
	NEXT record
ENDPROCEDURE

PROCEDURE AddClass()
	DECLARE className : STRING

	INPUT("Please enter class name", className)
	WHILE NOT IsValidName(className) DO
		INPUT("Invalid name, please enter again", className)
	ENDWHILE

	IF checkClassNameExist(className) THEN
		OUTPUT "Class already exists"
		RETURN
	ENDIF

	CALL DB_AddClass(className)
ENDPROCEDURE

PROCEDURE DeleteClass()
	DECLARE classID : INTEGER

	INPUT("Please enter class ID to delete", classID)
	WHILE NOT IsNumeric(classID) DO
		INPUT("Please enter a numeric class ID", classID)
	ENDWHILE

	CALL DB_DeleteClass(classID)
ENDPROCEDURE

PROCEDURE UpdateClass()
	DECLARE classID : INTEGER
	DECLARE newName : STRING

	INPUT("Updating, please enter class ID and new name", classID, newName)
	WHILE NOT IsNumeric(classID) OR NOT IsValidName(newName) DO
		INPUT("Invalid input, please enter class ID and new name", classID, newName)
	ENDWHILE

	CALL DB_UpdateClass(classID, newName)
ENDPROCEDURE

// ==============           DB Operation          =====================
PROCEDURE DB_AddClass(className : STRING)

	CONSTANT connection = ConnectToDatabase(database_url, credentials)
	IF connection == FALSE THEN
		OUTPUT "Connection error: " + connection.error
		RETURN
	ENDIF

	EXECUTE "INSERT INTO Class (className) VALUES (?)" WITH (className)
	OUTPUT "Class created"

	CALL CloseDatabaseConnection()

ENDPROCEDURE

PROCEDURE DB_DeleteClass(classID : INTEGER)
	DECLARE result : ARRAY OF DICTIONARY

	// Check BEFORE connecting
	IF NOT checkClassExist(classID) THEN
		OUTPUT "Class Not Found, Please enter again"
		RETURN
	ENDIF

	CONSTANT connection = ConnectToDatabase(database_url, credentials)
	IF connection == FALSE THEN
		OUTPUT "Connection error: " + connection.error
		RETURN
	ENDIF

	// Block deletion while students are still assigned to the class
	result = EXECUTE "SELECT COUNT(*) AS total FROM Student WHERE classID = ?" WITH (classID)
	IF result[0]["total"] > 0 THEN
		OUTPUT "Cannot delete: " + result[0]["total"] + " student(s) still in this class"
	ELSE
		EXECUTE "DELETE FROM Class WHERE classID = ?" WITH (classID)
		OUTPUT "Class deleted"
	ENDIF

	CALL CloseDatabaseConnection()

ENDPROCEDURE

PROCEDURE DB_UpdateClass(classID : INTEGER, newName : STRING)

	IF NOT checkClassExist(classID) THEN
		OUTPUT "Class Not Found"
		RETURN
	ENDIF

	IF checkClassNameExist(newName) THEN
		OUTPUT "A class with that name already exists"
		RETURN
	ENDIF

	CONSTANT connection = ConnectToDatabase(database_url, credentials)
	IF connection == FALSE THEN
		OUTPUT "Connection error: " + connection.error
		RETURN
	ENDIF

	EXECUTE "UPDATE Class SET className = ? WHERE classID = ?" WITH (newName, classID)
	OUTPUT "Successfully updated name to " + newName
	CALL CloseDatabaseConnection()

ENDPROCEDURE
